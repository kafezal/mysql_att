select 
    v.id_venda,
    date_format(v.data_venda, '%d/%m/%Y') as data_formatada,
    c.nome as cliente,
    c.cidade,
    p.nome as produto,
    iv.quantidade,
    iv.preco_unitario,
    v.valor_total
from vendas v
inner join clientes c on v.id_cliente = c.id_cliente
inner join itens_venda iv on v.id_venda = iv.id_venda
inner join produtos p on iv.id_produto = p.id_produto;
